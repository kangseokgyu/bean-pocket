import test from 'node:test';
import assert from 'node:assert/strict';
import {cp,mkdtemp,rm,readFile,writeFile,symlink} from 'node:fs/promises';
import os from 'node:os';
import path from 'node:path';
import {buildPages} from '/Users/kangseokgyu/workspace/AX-1day/scripts/build-pages.mjs';
import {checkPages} from '/Users/kangseokgyu/workspace/AX-1day/scripts/check-pages.mjs';
const root='/Users/kangseokgyu/workspace/AX-1day';
const modules=['commands/index.mjs','ui/controller.mjs','ui/view.mjs','platform/app-lifecycle.mjs','platform/update-coordinator.mjs'];
async function fixture(t){const dir=await mkdtemp(path.join(os.tmpdir(),'critic-build-boundary-'));t.after(()=>rm(dir,{recursive:true,force:true}));const sourceDir=path.join(dir,'pwa'),outputDir=path.join(dir,'dist');await cp(root+'/pwa',sourceDir,{recursive:true});return {dir,sourceDir,outputDir};}
for(const file of modules){
  test(`module round-trip/hash/cache and required output: ${file}`,async t=>{
    const f=await fixture(t);const v1=await buildPages(f),v2=await buildPages(f);assert.equal(v1,v2);
    assert.deepEqual(await readFile(path.join(f.sourceDir,file)),await readFile(path.join(f.outputDir,file)));
    const sw=await readFile(path.join(f.outputDir,'sw.js'),'utf8');assert.equal(JSON.parse(sw.match(/const APP_FILES\s*=\s*(\[[\s\S]*?\]);/)[1]).filter(s=>s==='./'+file).length,1);
    await writeFile(path.join(f.sourceDir,file),(await readFile(path.join(f.sourceDir,file),'utf8'))+'\n// independent hash check\n');
    assert.notEqual(await buildPages(f),v2);
    await rm(path.join(f.outputDir,file));await assert.rejects(checkPages(f.outputDir),/필수 산출물/);
  });
  test(`source leaf symlink: ${file}`,async t=>{
    const f=await fixture(t);const outside=path.join(f.dir,'outside.mjs');await writeFile(outside,'// sentinel\n');
    await rm(path.join(f.sourceDir,file));await symlink(outside,path.join(f.sourceDir,file));
    await assert.rejects(buildPages(f),/심볼릭|symlink/);assert.equal(await readFile(outside,'utf8'),'// sentinel\n');
  });
  test(`output leaf symlink rejects check and rebuild without modifying target: ${file}`,async t=>{
    const f=await fixture(t);await buildPages(f);const outside=path.join(f.dir,'outside.mjs');await writeFile(outside,'// sentinel\n');
    await rm(path.join(f.outputDir,file));await symlink(outside,path.join(f.outputDir,file));
    await assert.rejects(checkPages(f.outputDir),/심볼릭|symlink/);
    await assert.rejects(buildPages(f),/심볼릭|symlink/);assert.equal(await readFile(outside,'utf8'),'// sentinel\n');
  });
  test(`cache omission rejected: ${file}`,async t=>{
    const f=await fixture(t);await buildPages(f);const p=path.join(f.outputDir,'sw.js');const sw=await readFile(p,'utf8');
    await writeFile(p,sw.replace('"./'+file+'", ',''));await assert.rejects(checkPages(f.outputDir),/허용된 앱 파일/);
  });
}
for(const directory of ['commands','ui','platform']){
  test(`output directory symlink: ${directory}`,async t=>{
    const f=await fixture(t);await buildPages(f);const outside=path.join(f.dir,'outside');await cp(path.join(f.outputDir,directory),outside,{recursive:true});
    const sentinel=path.join(outside,'sentinel');await writeFile(sentinel,'preserve');
    await rm(path.join(f.outputDir,directory),{recursive:true});await symlink(outside,path.join(f.outputDir,directory));
    await assert.rejects(checkPages(f.outputDir),/심볼릭|symlink/);await assert.rejects(buildPages(f),/심볼릭|symlink/);
    assert.equal(await readFile(sentinel,'utf8'),'preserve');
  });
}
