<!doctype html><meta charset="utf-8"><title>Local product diagnostics</title><h1>Local product diagnostics</h1><button id="inspect">Inspect registration</button><button id="update">Check candidate update</button><pre id="result"></pre><script>
async function inspect(){const r=await navigator.serviceWorker.getRegistration('/candidate-resume/'); document.querySelector('#result').textContent=JSON.stringify({registration:r?{scope:r.scope,installing:r.installing?.state,waiting:r.waiting?.state,active:r.active?.state}:null,locks:await navigator.locks.query()},null,2);}
document.querySelector('#inspect').onclick=inspect; document.querySelector('#update').onclick=async()=>{const r=await navigator.serviceWorker.getRegistration('/candidate-resume/');await r?.update();await inspect();};inspect();
</script>
